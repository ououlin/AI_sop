### 2.1 数据库集成（SQLite）

**问题**：当前 SOP 仅存 JSON 文件，执行记录丢失，无法追溯历史数据。

**方案**：
- 引入 SQLite 数据库，轻量级无需额外服务
- 使用 SQLAlchemy ORM 管理数据库操作
- 创建三张表：SOP定义、执行记录、步骤记录
- 支持 SOP CRUD 操作和历史记录查询

**文件变更**：
- `backend/database.py`（新建）
- `backend/models/database.py`（新建）
- `backend/repositories/sop_repo.py`（新建）
- `backend/repositories/record_repo.py`（新建）
- `backend/api/v1/sop.py`（修改：使用数据库）
- `backend/api/v1/records.py`（新建）
- `backend/requirements.txt`（修改：添加 `sqlalchemy`）

**数据库模型**：
```python
# models/database.py
from sqlalchemy import Column, Integer, String, Float, DateTime, ForeignKey, Text
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import relationship

Base = declarative_base()

class SOPDefinitionDB(Base):
    __tablename__ = "sop_definitions"
    id = Column(Integer, primary_key=True, autoincrement=True)
    name = Column(String(100), unique=True, nullable=False)
    version = Column(String(20), default="1.0")
    steps_json = Column(Text, nullable=False)
    created_at = Column(DateTime)
    updated_at = Column(DateTime)

class ExecutionRecord(Base):
    __tablename__ = "execution_records"
    id = Column(Integer, primary_key=True, autoincrement=True)
    sop_id = Column(Integer, ForeignKey("sop_definitions.id"))
    sop_name = Column(String(100))
    start_time = Column(DateTime)
    end_time = Column(DateTime)
    status = Column(String(20))  # completed / interrupted / paused
    total_duration = Column(Float)
    
    steps = relationship("StepRecord", cascade="all, delete-orphan")

class StepRecord(Base):
    __tablename__ = "step_records"
    id = Column(Integer, primary_key=True, autoincrement=True)
    record_id = Column(Integer, ForeignKey("execution_records.id"))
    step_name = Column(String(100))
    step_index = Column(Integer)
    status = Column(String(20))  # pending / in_progress / completed / skipped
    duration = Column(Float)
    completed_at = Column(DateTime)
```

```python
@router.get("/mjpeg")
async def mjpeg_video():
    async def generate():
        while True:
            frame = camera.get_frame()
            if frame is None:
                await asyncio.sleep(0.03)
                continue
            ok, buf = cv2.imencode(".jpg", frame, [int(cv2.IMWRITE_JPEG_QUALITY), 75])
            yield f"--frame\r\nContent-Type: image/jpeg\r\n\r\n{buf.tobytes()}\r\n"
```

```python
# api/v1/camera.py
@router.get("/list")
async def list_cameras():
    cameras = []
    for i in range(5):
        cap = cv2.VideoCapture(i, cv2.CAP_MSMF)
        if cap.isOpened():
            w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
            h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
            cameras.append({"index": i, "resolution": f"{w}x{h}"})
            cap.release()
    return {"cameras": cameras}

@router.post("/switch")
async def switch_camera(index: int):
    camera.stop()
    camera.config.index = index
    success = camera.start()
    if success:
        return {"status": "success", "index": index}
    raise HTTPException(status_code=400, detail=f"无法打开摄像头 {index}")
```
